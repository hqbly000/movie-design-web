/* style.css —— 在 .ly-reveal 旁追加方向变体 */
.ly-reveal-x{
  opacity:.28;
  transform:translate3d(var(--reveal-x,0),var(--reveal-y,0),0);
  transition:opacity 620ms cubic-bezier(.22,.61,.36,1),
             transform 620ms cubic-bezier(.22,.61,.36,1);
  transition-delay:var(--reveal-delay,0ms);
  will-change:opacity,transform;
}
.ly-reveal-x.is-visible{opacity:1;transform:translate3d(0,0,0)}
@media (prefers-reduced-motion:reduce){
  .ly-reveal-x{opacity:1;transform:none;transition:none}
}
@media (max-width:767px){           /* 移动端纵向图带只保留淡入 */
  .ly-reveal-x{transform:none}
}

/* Segments.vue —— 给五列加指令（值 = 该列的起始位移与错峰毫秒） */



/* useInView.ts —— 新增 vRevealX 指令 */
export const vRevealX: Directive = {
  mounted(el, binding){
    const { x = 0, y = 0, delay = 0 } = binding.value ?? {}
    el.classList.add('ly-reveal-x')
    el.style.setProperty('--reveal-x', `${x}px`)
    el.style.setProperty('--reveal-y', `${y}px`)
    el.style.setProperty('--reveal-delay', `${delay}ms`)
    if (typeof IntersectionObserver === 'undefined'){
      el.classList.add('is-visible'); return
    }
    const io = new IntersectionObserver(([e]) => {
      if (e?.isIntersecting){ el.classList.add('is-visible'); io.disconnect() }
    }, { threshold:0.15, rootMargin:'0px 0px -8% 0px' })
    io.observe(el)
    ;(el as HTMLElement & {__ioX?:IntersectionObserver}).__ioX = io
  },
  unmounted(el){
    const h = el as HTMLElement & {__ioX?:IntersectionObserver}
    h.__ioX?.disconnect(); delete h.__ioX
  }
}

/* main.ts —— 注册 */
app.directive('reveal-x', vRevealX)
